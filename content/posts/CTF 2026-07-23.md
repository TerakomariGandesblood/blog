## 不会吧，就这？

```rust
use std::fs;

fn main() {
    let puzzle = fs::read_to_string("puzzle.txt").unwrap();
    let puzzle = puzzle
        .replace("不会吧？", "1")
        .replace("就这¿", "0")
        .replace(" ", "");

    for c in puzzle.lines() {
        let a = u32::from_str_radix(&c.chars().rev().collect::<String>(), 2).unwrap();
        print!("{}", char::from_u32(a).unwrap());
    }
}
```

## 想听点啥

使用 [MuseScore](https://musescore.org/zh-hans/download) 打开文件，获得压缩包密码：MOECTFI1iKE

```rust
use std::fs;

fn main() {
    let encrypt = fs::read("flag.txt").unwrap();
    let mut result = vec![b'm'];

    for (index, item) in encrypt.into_iter().enumerate() {
        result.push(item ^ result[index]);
    }

    let result: String = result.iter().map(|item| char::from(*item)).collect();
    println!("{result:?}");
}
```
